# Web Push Notifications Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add real browser push notifications so candidates get a system notification on application status changes (e.g., "Under review"), prompted once on login.

**Architecture:** Backend stores per-user `PushSubscription` rows and sends VAPID-signed pushes via the `web-push` package, fired best-effort from the existing `NotificationsService.createForUser` hook. Frontend registers a static service worker (`public/sw.js`), exposes `src/lib/push.ts` helpers, auto-prompts on login, and adds an enable affordance in the notification bell.

**Tech Stack:** Express 4 (ESM) + Prisma 5 + PostgreSQL (`web-push`, `@types/web-push`) · Vite 8 + React 19 + TS 6 · Vitest + supertest (backend), Vitest + Testing Library (frontend)

## Global Constraints

- Backend modules follow routes → controller → service + barrel `index.ts`, registered in `src/app/app.ts` (`app.use('/api', router)`).
- Auth via `requireAuth` (Bearer JWT); SSE uses `requireAuthQuery`. Zod validation via `validate(schema, source)` middleware (default `body`).
- Frontend API helpers in `src/api/client.ts` (`apiPost`, `apiDelete`). `VITE_*` env vars are auto-exposed by Vite.
- Push sending is ALWAYS best-effort: failures must never throw into notification creation (wrap in `.catch(() => {})` and log).
- No VAPID keys → graceful no-op with one log line (matches the existing Resend pattern in `src/services/email.ts`).
- Web Push needs a secure context: HTTPS in prod, `localhost` works in dev.
- Tests run against a real PostgreSQL DB and self-cleanup (follow `src/tests/notifications.test.ts`). Backend test command: `npm test` (vitest run). Frontend: `npm run test:run`.
- Verify each task with the repo's own commands: backend `npm run build` (`tsc`) + `npm test`; frontend `npm run build` (`tsc -b && vite build`) + `npm run lint` + `npm run test:run`.
- Backend logger: `import { logger } from '../../config/logger'`.

Create `src/modules/push/push.service.ts`:

```ts
import webpush from 'web-push'
import { prisma } from '../../lib/prisma'
import { env } from '../../config/env'
import { logger } from '../../config/logger'

if (env.VAPID_PUBLIC_KEY && env.VAPID_PRIVATE_KEY) {
  webpush.setVapidDetails(env.VAPID_SUBJECT ?? 'mailto:hello@hirehub.com', env.VAPID_PUBLIC_KEY, env.VAPID_PRIVATE_KEY)
}

export class PushService {
  async subscribe(userId: string, input: { endpoint: string; p256dh: string; auth: string }) {
    return prisma.pushSubscription.upsert({
      where: { endpoint: input.endpoint },
      create: { userId, ...input },
      update: { userId, p256dh: input.p256dh, auth: input.auth },
    })
  }

  async unsubscribe(userId: string, endpoint: string) {
    await prisma.pushSubscription.deleteMany({ where: { userId, endpoint } })
    return { deleted: true }
  }

  async sendPush(userId: string, payload: { title: string; body: string; url: string }) {
    if (!env.VAPID_PRIVATE_KEY) {
      logger.info('Push disabled: VAPID_PRIVATE_KEY not configured')
      return
    }

    const subscriptions = await prisma.pushSubscription.findMany({ where: { userId } })
    if (subscriptions.length === 0) return

    const body = JSON.stringify(payload)

    await Promise.allSettled(
      subscriptions.map(async (sub) => {
        try {
          await webpush.sendNotification(
            { endpoint: sub.endpoint, keys: { p256dh: sub.p256dh, auth: sub.auth } },
            body,
            { TTL: 86400 },
          )
        } catch (err) {
          const statusCode = (err as { statusCode?: number }).statusCode
          if (statusCode === 404 || statusCode === 410) {
            await prisma.pushSubscription.deleteMany({ where: { id: sub.id } })
          } else {
            logger.warn({ err, endpoint: sub.endpoint }, 'Push send failed')
          }
        }
      }),
    )
  }
}

export const pushService = new PushService()
```

```js
self.addEventListener('install', () => {
  self.skipWaiting()
})

self.addEventListener('activate', (event) => {
  event.waitUntil(clients.claim())
})

self.addEventListener('push', (event) => {
  let data = { title: 'HireHub', body: '', url: '/dashboard' }
  try {
    data = event.data ? event.data.json() : data
  } catch {
    // fall back to defaults on malformed payloads
  }
  event.waitUntil(
    self.registration.showNotification(data.title, {
      body: data.body,
      icon: '/favicon.svg',
      badge: '/favicon.svg',
      data: { url: data.url || '/dashboard' },
    }),
  )
})

self.addEventListener('notificationclick', (event) => {
  event.notification.close()
  const url = (event.notification.data && event.notification.data.url) || '/dashboard'
  event.waitUntil(
    clients.matchAll({ type: 'window', includeUncontrolled: true }).then((windowClients) => {
      for (const client of windowClients) {
        if ('focus' in client) {
          client.navigate(url)
          return client.focus()
        }
      }
      return clients.openWindow(url)
    }),
  )
})
```

```ts
import { apiPost, apiDelete } from '../api/client'

const SW_PATH = '/sw.js'

export function isPushSupported(): boolean {
  return typeof window !== 'undefined' && 'serviceWorker' in navigator && 'PushManager' in window
}

export function getVapidPublicKey(): string | undefined {
  const key = (import.meta as unknown as { env?: Record<string, string> }).env?.VITE_VAPID_PUBLIC_KEY
  return key || undefined
}

export function urlBase64ToUint8Array(base64String: string): Uint8Array {
  const padding = '='.repeat((4 - (base64String.length % 4)) % 4)
  const base64 = (base64String + padding).replace(/-/g, '+').replace(/_/g, '/')
  const rawData = window.atob(base64)
  const outputArray = new Uint8Array(rawData.length)
  for (let i = 0; i < rawData.length; i++) outputArray[i] = rawData.charCodeAt(i)
  return outputArray
}

export async function registerPushServiceWorker(): Promise<ServiceWorkerRegistration | undefined> {
  if (!isPushSupported()) return undefined
  return navigator.serviceWorker.register(SW_PATH)
}

export async function subscribeUserToPush(): Promise<boolean> {
  if (!isPushSupported()) return false
  if (typeof Notification !== 'undefined' && Notification.permission !== 'granted') {
    const permission = await Notification.requestPermission()
    if (permission !== 'granted') return false
  }
  const vapidKey = getVapidPublicKey()
  if (!vapidKey) return false

  const registration = await registerPushServiceWorker()
  if (!registration) return false

  let subscription = await registration.pushManager.getSubscription()
  if (!subscription) {
    subscription = await registration.pushManager.subscribe({
      userVisibleOnly: true,
      applicationServerKey: urlBase64ToUint8Array(vapidKey),
    })
  }

  const p256dh = subscription.getKey('p256dh')
  const auth = subscription.getKey('auth')
  if (!p256dh || !auth) return false

  await apiPost('/push/subscribe', {
    endpoint: subscription.endpoint,
    p256dh: btoa(String.fromCharCode(...new Uint8Array(p256dh))),
    auth: btoa(String.fromCharCode(...new Uint8Array(auth))),
  })
  return true
}

export async function unsubscribeUserFromPush(): Promise<void> {
  if (!isPushSupported()) return
  const registration = await navigator.serviceWorker.getRegistration(SW_PATH)
  const subscription = await registration?.pushManager.getSubscription()
  if (!subscription) return
  await subscription.unsubscribe()
  await apiDelete(`/push/unsubscribe?endpoint=${encodeURIComponent(subscription.endpoint)}`)
}

export async function enablePushIfUndecided(): Promise<void> {
  if (!isPushSupported()) return
  if (typeof Notification !== 'undefined' && Notification.permission === 'default') {
    try {
      await subscribeUserToPush()
    } catch (err) {
      console.info('[push] enabling browser notifications skipped', err)
    }
  }
}
```
